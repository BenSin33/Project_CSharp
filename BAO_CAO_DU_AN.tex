\chapter{Tổng quan dự án}

\section{Mô tả dự án}
InteractHub là một nền tảng mạng xã hội được xây dựng với công nghệ hiện đại, cho phép người dùng:
\begin{itemize}
    \item Tạo, sửa, xóa các bài viết (Post)
    \item Tương tác với bài viết (Like, Comment, Share)
    \item Quản lý bạn bè (Friendship)
    \item Chia sẻ story
    \item Gửi thông báo (Notification)
    \item Xem và quản lý các hashtag
    \item Báo cáo bài viết không phù hợp
\end{itemize}

\section{Chi tiết từng requirement}

\subsection{Requirement B1: Database Design \& Entity Framework}
\textbf{Trạng thái: } \textcolor{darkblue}{\textbf{85-90\% Hoàn thành}}

\subsubsection{Đã hoàn thành}
\begin{itemize}
    \item[$\checkmark$] Thiết kế 9 Entity:
    \begin{itemize}
        \item User, Post, Comment, Like, Share
        \item FriendShip, Story, Notification, HashTag, PostReport
    \end{itemize}
    \item[$\checkmark$] Lớp BaseEntity với Guid, CreatedAt, UpdatedAt, DeletedAt
    \item[$\checkmark$] ApplicationDbContext được cấu hình đúng
    \item[$\checkmark$] Quan hệ One-to-Many, One-to-One, Many-to-Many
    \item[$\checkmark$] Fluent API Configuration
    \item[$\checkmark$] Cascade delete rules
    \item[$\checkmark$] Seeding dữ liệu test
    \item[$\checkmark$] Migration file (20260413090827\_InitialCreate.cs)
\end{itemize}

\subsubsection{Vấn đề và khuyến nghị}
\begin{enumerate}
    \item \textbf{Cần thêm 2-3 migration files} (yêu cầu: ít nhất 3)
    \begin{itemize}
        \item Hiện có: 1 migration chính
        \item Đề xuất: Thêm migration cho schema refinements, constraints, new columns
    \end{itemize}
    
    \item \textbf{Thiếu Data Annotations}
    \begin{itemize}
        \item Cần thêm: [Required], [StringLength], [EmailAddress], etc.
        \item Ví dụ: User.FullName cần [Required]
    \end{itemize}
    
    \item \textbf{Thiếu Unique Constraints}
    \begin{itemize}
        \item Like entity: (UserId, PostId) nên là unique
        \item FriendShip: (RequesterId, ReceiverId) nên là unique
    \end{itemize}
\end{enumerate}
